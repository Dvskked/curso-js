Modulo 01

Fundamentos de JavaScript

En este modulo aprenderas los cimientos de todo programa en JavaScript. Comenzaremos con las variables, los tipos de datos y las operaciones basicas que son la base para todo lo demas que aprenderas en este curso.

1.1 Variables: var, let y const

JavaScript ofrece tres formas de declarar variables: var, let y const. Cada una tiene un comportamiento diferente en cuanto a alcance, elevacion (hoisting) y reasignacion.

La variable var (legado)

var es la forma antigua. Tiene alcance de funcion, accesible desde cualquier lugar dentro de la funcion donde se declaro. 

// var: alcance de funcion (no recomendado en codigo moderno)
var nombre = "Ana";
var edad = 25;
var activo = true;

// Problema con var: se escapa del bloque
if (true) {
  var mensaje = "Hola";
}
console.log(mensaje); // "Hola" - var no respeta el bloque

La variable let (recomendada)

let tiene alcance de bloque. Solo es accesible dentro del bloque {} donde se declaro. Ideal para valores que cambian. 

// let: alcance de bloque (recomendado para valores que cambian)
let contador = 0;
let nombreUsuario = "Carlos";

if (true) {
  let temporal = "Solo existe aqui";
  console.log(temporal); // "Solo existe aqui"
}
// console.log(temporal); // ReferenceError

// let permite reasignacion
contador = 1;
contador = 2; // Perfectamente valido

La constante const (recomendada)

const tiene alcance de bloque pero su valor no puede ser reasignado. Es la forma recomendada por defecto. 

// const: no reasignable (recomendado por defecto)
const PI = 3.14159;
const API_URL = "https://api.ejemplo.com";
const MAX_INTENTOS = 3;

// PI = 3.14; // TypeError: Assignment to constant variable.

// Importante: const no significa "inmutable"
const usuario = { nombre: "Luis", edad: 30 };
usuario.edad = 31; // Se puede modificar propiedades
// usuario = {}; // Pero no se puede reasignar la referencia

✗ Uso incorrecto de var

for (var i = 0; i < 3; i++) {
  setTimeout(() => {
    console.log(i); // Imprime 3, 3, 3
  }, 1000);
}
// var tiene alcance de funcion, no de bloque

✓ Uso correcto de let

for (let i = 0; i < 3; i++) {
  setTimeout(() => {
    console.log(i); // Imprime 0, 1, 2
  }, 1000);
}
// let crea una nueva variable en cada iteracion
Regla de oro: Usa const por defecto. Cambia a let solo cuando necesites reasignar. Nunca uses var en codigo moderno.
Ejercicio Practico: Declara variables con const para tu nombre, apellido y edad. Intenta reasignar cada una y observa los errores. Luego crea una variable let para un contador y aumentala de 1 en 1 hasta llegar a 10.

1.2 Tipos de Datos en JavaScript

JavaScript es un lenguaje de tipado dinamico. El tipo se determina en tiempo de ejecucion. Tiene 8 tipos, divididos en primitivos y no primitivos.

Tipos Primitivos

// Los tipos primitivos de JavaScript
let texto = "Hola Mundo";           // string
let entero = 42;                    // number
let decimal = 3.14;                  // number
let verdadero = true;                // boolean
let nada = null;                    // null
let indefinido;                       // undefined
let id = 9007199254740991n;          // bigint
let simbolo = Symbol("id");         // symbol

El operador typeof

// typeof devuelve el tipo como string
console.log(typeof "Hola");       // "string"
console.log(typeof 42);          // "number"
console.log(typeof true);        // "boolean"
console.log(typeof undefined);   // "undefined"
console.log(typeof null);        // "object" (bug historico!)
console.log(typeof [1, 2, 3]);    // "object"
console.log(typeof {});          // "object"
Cuidado: typeof null devuelve "object" en lugar de "null". Es un bug historico que nunca se corrigio.

Diferencia entre undefined y null

// undefined: asignado automaticamente por JavaScript
let edad;
console.log(edad);           // undefined
console.log(typeof edad);  // "undefined"

// null: ausencia intencional de valor
let resultado = null;
console.log(resultado);      // null

// Funcion que busca un usuario
function buscarUsuario(id) {
  const usuario = usuarios.find(u => u.id === id);
  return usuario || null; // null si no encontro
}

✗ Confusion con tipos

let precio = "100";  // Es string!
let total = precio + 50;
console.log(total);     // "10050"

✓ Uso correcto

let precio = 100;  // number
let total = precio + 50;
console.log(total);     // 150

1.3 Operadores en JavaScript

Los operadores son simbolos que realizan operaciones sobre valores. JavaScript tiene aritmeticos, asignacion, comparacion y logicos.

Operadores Aritmeticos

let a = 10;
let b = 3;

console.log(a + b);   // 13    (suma)
console.log(a - b);   // 7     (resta)
console.log(a * b);   // 30    (multiplicacion)
console.log(a / b);   // 3.333 (division)
console.log(a % b);   // 1     (modulo)
console.log(a ** b);  // 1000  (potencia)

let contador = 5;
contador++;       // 6
contador--;       // 5
let nuevo = ++contador; // pre-incremento

Operadores de Comparacion

// == vs === (abstracta vs estricta)
console.log(5 == "5");     // true  (convierte tipos)
console.log(5 === "5");    // false (mismos tipos)
console.log(0 == "");      // true  (inesperado)
console.log(0 === "");     // false (correcto)

// != vs !==
console.log(5 != "5");     // false
console.log(5 !== "5");    // true
Siempre usa === (igualdad estricta): == convierte tipos antes de comparar, causando resultados inesperados.

Operadores Logicos

// AND (&&): ambas deben ser true
console.log(true && true);    // true
console.log(true && false);   // false

// OR (||): al menos una true
console.log(true || false);   // true

// NOT (!): invierte
console.log(!true);            // false

// Valores por defecto con ||
const nombre = "" || "Visitante";
console.log(nombre);  // "Visitante"

✗ Usando ==

let edad = "0";
if (edad == 0) {
  console.log("Sin edad");
}
// Se ejecuta porque "0" == 0

✓ Usando ===

let edad = "0";
if (edad === 0) {
  console.log("Sin edad");
}
// No se ejecuta: string !== number
Ejercicio: Resuelve: true && "hello", false || 42, !"", 0 || "fallback". Escribe un programa que determine si un numero es par.

1.4 Strings: Manipulacion de Texto

Los strings se crean con comillas simples, dobles o backticks. Tienen metodos incorporados para manipular texto.

Creacion de Strings

// Tres formas de crear strings
const str1 = "Hola con comillas dobles";
const str2 = 'Hola con comillas simples';
const str3 = `Hola con backticks`;

// Backticks: interpolacion y multilínea
const nombre = "Mundo";
const saludo = `Hola ${nombre}!`;
console.log(saludo);  // "Hola Mundo!"

Metodos de String

const texto = "JavaScript es increible";

console.log(texto.length);           // 22
console.log(texto.toUpperCase());    // "JAVASCRIPT ES INCREIBLE"
console.log(texto.toLowerCase());    // "javascript es increible"
console.log(texto.includes("Script"));  // true
console.log(texto.indexOf("es"));      // 11
console.log(texto.startsWith("Java")); // true
console.log(texto.endsWith("ble"));   // true
console.log(texto.slice(0, 10));      // "JavaScript"
console.log(texto.slice(-9));          // "increible"

Busqueda y Reemplazo

const mensaje = "Me gusta Python y JavaScript";

console.log(mensaje.replace("Python", "JS"));
// "Me gusta JS y JavaScript"

const todos = "aabbcc".replaceAll("a", "x");
console.log(todos);  // "xxbbcc"

const csv = "manzana,pera,banana";
const frutas = csv.split(",");
console.log(frutas);  // ["manzana", "pera", "banana"]

const unido = frutas.join(" - ");
console.log(unido);   // "manzana - pera - banana"
Inmutabilidad: Los strings son inmutables. toUpperCase() devuelve un nuevo string, no modifica el original.

1.5 Numbers y Operaciones Numericas

Todos los numeros en JavaScript son del tipo number (IEEE 754 doble precision). No hay distincion entre int y float.

Valores Especiales

const inf = Infinity;       // Infinity
const nInf = -Infinity;      // -Infinity
const nan = NaN;              // NaN (Not a Number)

// NaN aparece con operaciones invalidas
console.log("hola" * 2);      // NaN
console.log(0 / 0);            // NaN

// Verificar NaN
console.log(Number.isNaN(NaN));  // true
console.log(NaN === NaN);            // false (por diseno)

Conversion de Numeros

console.log(Number("42"));        // 42
console.log(Number("3.14"));      // 3.14
console.log(Number(""));           // 0
console.log(Number("hola"));       // NaN

console.log(parseInt("42.9"));     // 42 (trunca)
console.log(parseInt("100px"));     // 100
console.log(parseFloat("3.14abc")); // 3.14

console.log(+"42");    // 42 (operador unary)

Metodos de Numero

console.log(Math.round(4.5));     // 5
console.log(Math.ceil(4.1));      // 5 (hacia arriba)
console.log(Math.floor(4.9));     // 4 (hacia abajo)

console.log((19.995).toFixed(2));  // "20.00" (string!)

console.log(Math.random());        // 0.0 - 1.0
const dado = Math.floor(Math.random() * 6) + 1;
console.log(dado);                 // 1 - 6

console.log(Math.min(5, 3, 8, 1));  // 1
console.log(Math.max(5, 3, 8, 1));  // 8
Precision: 0.1 + 0.2 no es exactamente 0.3. El resultado es 0.30000000000000004 por IEEE 754.

1.6 Booleans, null y undefined

Los booleans solo tienen true y false. JavaScript tiene truthy y falsy.

Valores Truthy y Falsy

// Solo estos 8 valores son falsy
console.log(Boolean(false));       // false
console.log(Boolean(0));             // false
console.log(Boolean(0n));            // false (BigInt cero)
console.log(Boolean(""));            // false (string vacio)
console.log(Boolean(null));          // false
console.log(Boolean(undefined));     // false
console.log(Boolean(NaN));            // false

// Todo lo demas es truthy
console.log(Boolean("0"));    // true
console.log(Boolean([]));     // true
console.log(Boolean({}));     // true

null vs undefined

// undefined: asignado automaticamente
let nombre;
console.log(nombre);           // undefined
console.log(typeof nombre);  // "undefined"

// null: ausencia intencional de valor
let usuario = null;
console.log(usuario);          // null
console.log(typeof usuario);   // "object" (bug)

// Verificar correctamente
console.log(usuario === null); // true

✗ Confusion truthy/falsy

const edad = "0";
if (edad) {
  console.log("Tiene edad");
}
// "0" es truthy!

✓ Comparacion explicita

const edad = "0";
if (edad !== undefined && edad !== "") {
  console.log("Tiene edad");
}

1.7 Type Coercion: Conversion Automatica

Type coercion es la conversion implicita de tipos que JavaScript hace durante operaciones.

Coercion Explicita vs Implicita

// EXPLICITA: tu haces la conversion
const num = Number("42");    // 42
const str = String(42);        // "42"
const bool = Boolean(1);       // true

// IMPLICITA: JavaScript lo hace solo
console.log("5" - 2);     // 3  (string a number)
console.log("5" + 2);     // "52" (concatenacion!)
console.log("5" * 2);     // 10 (string a number)
El + es especial: Con un string, + concatena. "5" + 2 es "52", no 7.

Coercion en Comparaciones

// == convierte tipos
console.log(1 == "1");       // true
console.log(0 == "");        // true
console.log(0 == false);     // true

// === NO convierte
console.log(1 === "1");      // false
console.log(0 === "");       // false
console.log(0 === false);    // false

✗ Inesperados con ==

console.log("" == false);   // true
console.log("0" == false);  // true
console.log([] == false);   // true

✓ Seguros con ===

console.log("" === false);  // false
console.log("0" === false); // false
console.log([] === false);  // false

Tabla de Conversion a Number

console.log(Number("42"));          // 42
console.log(Number(""));             // 0
console.log(Number(true));           // 1
console.log(Number(false));          // 0
console.log(Number(null));           // 0
console.log(Number(undefined));      // NaN

Estructuras de Control

Las estructuras de control determinan el flujo de ejecucion de tu programa. Aprenderas a tomar decisiones con condicionales y a repetir acciones con bucles.

2.1 Condicionales: if, else if, else

La sentencia if ejecuta un bloque de codigo si una condicion es truthy. Puedes encadenar else if y else para manejar multiples caminos.

Estructura Basica

const edad = 18;

if (edad >= 18) {
  console.log("Eres mayor de edad");
} else {
  console.log("Eres menor de edad");
}
// "Eres mayor de edad"

Encadenar multiples condiciones

const nota = 75;

if (nota >= 90) {
  console.log("Excelente");
} else if (nota >= 70) {
  console.log("Bueno");
} else if (nota >= 50) {
  console.log("Regular");
} else {
  console.log("Reprobado");
}
// "Bueno"

Combinar condiciones con && y ||

const usuario = { activo: true, admin: false };

if (usuario.activo && usuario.admin) {
  console.log("Admin activo");
} else if (usuario.activo || usuario.admin) {
  console.log("Activo o admin");
} else {
  console.log("Sin acceso");
}
// "Activo o admin"
Cuidado con la negacion: !usuario.activo && !usuario.admin es dificil de leer. Prefiere !(usuario.activo || usuario.admin).

✗ Asignacion accidental

const x = 0;
if (x = 5) {  // Asignacion, no comparacion!
  console.log("Siempre ejecuta");
}

✓ Comparacion estricta

const x = 0;
if (x === 5) {
  console.log("Solo si x es 5");
}

2.2 Sentencia switch

switch evalua una expresion y ejecuta el caso que coincida. Cada caso requiere break.

Uso Basico

const dia = "lunes";

switch (dia) {
  case "lunes":
  case "martes":
  case "miercoles":
  case "jueves":
  case "viernes":
    console.log("Dia laboral");
    break;
  case "sabado":
  case "domingo":
    console.log("Fin de semana");
    break;
  default:
    console.log("Dia invalido");
}
No olvides break: Sin break, el switch "cae" al siguiente caso (fall-through). A veces util, pero usualmente un error.

Switch con expresiones

const hora = 14;

switch (true) {
  case (hora < 12):
    console.log("Buenos dias");
    break;
  case (hora < 18):
    console.log("Buenas tardes");
    break;
  default:
    console.log("Buenas noches");
}

✗ Sin break

const tipo = "admin";
switch (tipo) {
  case "admin":
    console.log("Admin");
  case "user":   // fall-through!
    console.log("User");
}

✓ Con break

const tipo = "admin";
switch (tipo) {
  case "admin":
    console.log("Admin");
    break;
  case "user":
    console.log("User");
    break;
}

2.3 Operador Ternario

El operador ternario es un if/else en una linea: condicion ? valorSiTrue : valorSiFalse.

Sintaxis basica

const edad = 20;
const estado = edad >= 18 ? "adulto" : "menor";
console.log(estado);  // "adulto"

// Es equivalente a:
let estado2;
if (edad >= 18) {
  estado2 = "adulto";
} else {
  estado2 = "menor";
}

Ternarios anidados

const nota = 85;
const resultado = nota >= 90 ? "A"
  : nota >= 80 ? "B"
  : nota >= 70 ? "C"
  : "F";
console.log(resultado);  // "B"

✗ Ternarios complejos

// Dificil de leer
const x = a > b ? (c > d ? e : f) : (g ? h : i);

✓ if/else para logica compleja

// Mejor: if/else para logica compleja
let x;
if (a > b) {
  x = c > d ? e : f;
} else {
  x = g ? h : i;
}
Regla: Usa ternarios para decisiones simples en una linea. Para logica compleja, usa if/else.
Ejercicio: Escribe saludo(nombre, hora) con ternario que retorne "Buenos dias", "Buenas tardes" o "Buenas noches".

2.4 Bucle for clasico

El bucle for repite un bloque un numero determinado de veces.

Estructura del for

// for (inicio; condicion; incremento)
for (let i = 0; i < 5; i++) {
  console.log(i);
}
// 0, 1, 2, 3, 4

Patrones comunes

// Contar hacia atras
for (let i = 10; i >= 0; i -= 2) {
  console.log(i);  // 10, 8, 6, 4, 2, 0
}

// Multiplicar
for (let i = 1; i <= 10; i++) {
  console.log(`5 x ${i} = ${5 * i}`);
}
Evita var en for: Usa let para evitar problemas de alcance con closures y callbacks como setTimeout.
Ejercicio: Escribe un bucle que imprima todos los numeros pares del 1 al 50. Luego uno que imprima los multiplos de 3.

2.5 Bucles while y do-while

Los bucles while y do-while repiten mientras la condicion sea truthy.

while basico

let contador = 0;
while (contador < 5) {
  console.log(contador);
  contador++;
}
// 0, 1, 2, 3, 4

do-while: ejecuta al menos una vez

let intento = 0;
do {
  intento++;
  console.log(`Intento ${intento}`);
} while (intento < 3);

Ejemplo practico: juego

const secreto = 42;
let intentos = 0;
let adivinado = false;

while (!adivinado && intentos < 3) {
  intentos++;
  const intento = Math.floor(Math.random() * 100);
  if (intento === secreto) {
    adivinado = true;
  }
}
Cuidado con el bucle infinito: Si la condicion nunca es falsa, el programa se cuelga. Siempre asegurate que la variable de control cambie.

✗ Bucle infinito

let i = 0;
while (i < 5) {
  console.log(i);
  // Olvide incrementar i!
}

✓ Incremento correcto

let i = 0;
while (i < 5) {
  console.log(i);
  i++;  // Siempre incrementa
}

Funciones

Las funciones son bloques de codigo reutilizables. En este modulo aprenderas a crearlas, pasarles parametros, entender closures y usarlas como callbacks.

3.1 Declaracion de Funciones

Una funcion es un bloque de codigo que ejecuta una tarea. Se define con function y se invoca con ().

Sintaxis basica

// Declaracion de funcion
function saludar() {
  console.log("Hola, mundo!");
}

saludar();  // Puedes llamarla muchas veces

Parametros y retorno

function sumar(a, b) {
  return a + b;
}

const resultado = sumar(5, 3);
console.log(resultado);  // 8

// Sin return, la funcion retorna undefined
function saludar2(nombre) {
  console.log(`Hola ${nombre}`);
}
const res = saludar2("Ana");
console.log(res);  // undefined

Hoisting (elevacion)

// Las declaraciones se "elevan" al inicio del scope

function miFuncion() {
  console.log("Funciona por hoisting");
}

✗ Expresion con var

var cuadrado = function(x) {
  return x * x;
};

✓ Declaracion de funcion

function cuadrado(x) {
  return x * x;
}

3.2 Expresiones de Funcion

Una expresion de funcion asigna una funcion a una variable. No tiene hoisting.

Sintaxis

// Expresion anonima
const sumar = function(a, b) {
  return a + b;
};

// Expresion con nombre
const restar = function restar(a, b) {
  return a - b;
};

console.log(sumar(5, 3));  // 8

Hoisting: la diferencia clave

// Declaracion: funciona antes de definirla
suma(2, 3);  // 5
function suma(a, b) { return a + b; }

// Expresion: NO funciona antes
resta(5, 2);  // ReferenceError!
const resta = function(a, b) { return a - b; };

Uso en callbacks

// Expresiones son utiles como callbacks
[1, 2, 3].map(function(num) {
  return num * 2;
});

// Funcion que recibe otra funcion
function operar(a, b, operacion) {
  return operacion(a, b);
}
const suma = operar(5, 3, function(a, b) { return a + b; });
console.log(suma);  // 8
Diferencia con arrow: Las expresiones de funcion son utiles cuando necesitas el contexto this dinamico o arguments.

3.3 Arrow Functions (ES6)

Las arrow functions son una forma compacta de escribir funciones. Introducidas en ES6.

Sintaxis basica

// Basico
const saludar = (nombre) => {
  return `Hola ${nombre}!`;
};

// Conciso: una sola expresion
const doblar = n => n * 2;

// Un solo parametro: parentesis opcionales
const esPositivo = n => n > 0;

// Sin parametros: parentesis obligatorio
const hola = () => "Hola!";

// Multiples parametros
const sumar = (a, b) => a + b;

Retorno implicito y explicito

// Explicito: con return y llaves
const cuadrado = (x) => {
  return x * x;
};

// Implicito: sin return ni llaves
const cuadrado2 = x => x * x;

// Retornar objeto literal: parentesis obligatorio
const crearUsuario = (nombre, edad) => ({
  nombre,
  edad,
  activo: true
});

Diferencia con this

// Arrow: hereda this del scope externo
const contador = {
  valor: 0,
  incrementar: () => {
    // this es el scope externo, NO contador
    // this.valor es undefined
  }
};

// Funcion regular: this es el objeto
const contador2 = {
  valor: 0,
  incrementar() {
    this.valor++;  // this es contador2
  }
};

✗ Function expression

// Muchos caracteres
[1,2,3].map(function(x) {
  return x * 2;
});

✓ Arrow function

// Conciso y claro
[1,2,3].map(x => x * 2);
No uses arrow functions: Como metodos de objeto, como constructores, o cuando necesitas this dinamico.

3.4 Parametros default, rest y spread

Valores por Defecto

  return `Hola ${nombre}!`;
}

  return `${prefijo}-${Math.floor(num * 1000)}`;
}

// Recibe argumentos ilimitados como array

console.log(sumar(1, 2, 3, 4, 5)); // 15

// Spread: expandir array como argumentos

3.6 Closures

Un closure es cuando una funcion "recuerda" las variables de su scope exterior, incluso despues de que esa funcion haya terminado de ejecutarse.

Ejemplo basico de closure

function creadorSaludo(saludo) {
  return function(nombre) {
    return `${saludo}, ${nombre}!`;
  };
}

const saludarEn = creadorSaludo("Hola");
const saludarEnIngles = creadorSaludo("Hello");

console.log(saludarEn("Carlos"));      // "Hola, Carlos!"
console.log(saludarEnIngles("John"));  // "Hello, John!"

Contador con closure

function crearContador() {
  let cuenta = 0;
  return {
    incrementar() { return ++cuenta; },
    decrementar() { return --cuenta; },
    obtener() { return cuenta; }
  };
}

const contador = crearContador();
console.log(contador.incrementar());  // 1
console.log(contador.incrementar());  // 2
console.log(contador.obtener());     // 2

Closure en loops (problema clasico)

// Problema clasico con var
for (var i = 0; i < 3; i++) {
  setTimeout(() => console.log(i), 1000);
}
// 3, 3, 3 (closure captura var)

// Solucion 1: let
for (let i = 0; i < 3; i++) {
  setTimeout(() => console.log(i), 1000);
}
// 0, 1, 2

// Solucion 2: IIFE con closure
for (var i = 0; i < 3; i++) {
  (function(j) {
    setTimeout(() => console.log(j), 1000);
  })(i);
}
Closure: Toda funcion en JavaScript es un closure. Puede acceder a variables de su scope exterior.

Un callback es una funcion que se pasa como argumento a otra funcion para ser ejecutada despues.

  console.log("Despues de 2s");

const numeros = [1, 2, 3, 4, 5];

const result = [];

Domina las estructuras de datos mas importantes en JavaScript: arrays para colecciones ordenadas y objetos para modelar entidades con propiedades.

4.1 Creacion y Propiedades de Arrays

Los arrays son colecciones ordenadas de valores. En JavaScript pueden contener cualquier tipo de dato y no tienen un tamano fijo.

Crear Arrays

Existen varias formas de crear arrays en JavaScript:

// Literales de array
const numeros = [1, 2, 3, 4, 5];
const frutas = ["manzana", "banana", "naranja"];
const mixto = [1, "texto", true, null];
// Constructor de Array
const arrVacio = new Array();
const arrTamano = new Array(5); // Array de 5 elementos vacios
// Array.of crea un array con los argumentos
const arrOf = Array.of(1, 2, 3); // [1, 2, 3]

Metodos Basicos: push, pop, shift, unshift

Estos metodos modifican el array directamente (mutacion):

const pila = ["a", "b", "c"];
// push: agrega al final
pila.push("d");      // ["a", "b", "c", "d"]
// pop: elimina del final
const ultimo = pila.pop(); // "d" -- pila: ["a", "b", "c"]
// unshift: agrega al inicio
pila.unshift("z");  // ["z", "a", "b", "c"]
// shift: elimina del inicio
const primero = pila.shift(); // "z" -- pila: ["a", "b", "c"]

length y indexOf

const colores = ["rojo", "verde", "azul"];
// length: cantidad de elementos
console.log(colores.length); // 3
// indexOf: posicion de un elemento (-1 si no existe)
console.log(colores.indexOf("verde"));  // 1
console.log(colores.indexOf("amarillo")); // -1
Cuidado: push, pop, shift y unshift modifican el array original. Si necesitas preservar el original, crea una copia primero con slice() o el spread operator.

4.2 Metodos de Iteracion

JavaScript ofrece poderosos metodos para recorrer y transformar arrays sin usar bucles for explicitos.

forEach y map

const numeros = [1, 2, 3, 4, 5];
// forEach: ejecuta una funcion por cada elemento (no retorna nada)
numeros.forEach(function(num, indice) {
  console.log(`Indice ${indice}: ${num}`);
});
// map: crea un nuevo array aplicando una funcion a cada elemento
const dobles = numeros.map(num => num * 2);
console.log(dobles); // [2, 4, 6, 8, 10]
// map con array de objetos
const usuarios = [
  { nombre: "Ana", edad: 25 },
  { nombre: "Luis", edad: 30 }
];
const nombres = usuarios.map(u => u.nombre);
console.log(nombres); // ["Ana", "Luis"]

filter y reduce

const edades = [15, 22, 18, 30, 12];
// filter: crea un nuevo array con elementos que cumplen una condicion
const mayores = edades.filter(edad => edad >= 18);
console.log(mayores); // [22, 18, 30]
// reduce: acumula los elementos en un solo valor
const suma = edades.reduce((acum, actual) => acum + actual, 0);
console.log(suma); // 97
// reduce para contar ocurrencias
const letras = ["a", "b", "a", "c", "a", "b"];
const conteo = letras.reduce((acc, letra) => {
  acc[letra] = (acc[letra] || 0) + 1;
  return acc;
}, {});
console.log(conteo); // {a: 3, b: 2, c: 1}

find, findIndex, some y every

const productos = [
  { id: 1, nombre: "Laptop", precio: 999 },
  { id: 2, nombre: "Mouse", precio: 25 },
  { id: 3, nombre: "Teclado", precio: 75 }
];
// find: retorna el primer elemento que cumple la condicion
const caro = productos.find(p => p.precio > 100);
console.log(caro); // {id: 1, nombre: "Laptop", precio: 999}
// findIndex: retorna el indice del primer elemento encontrado
const idx = productos.findIndex(p => p.nombre === "Mouse");
console.log(idx); // 1
// some: retorna true si AL MENOS UNO cumple la condicion
const hayBarato = productos.some(p => p.precio < 50);
console.log(hayBarato); // true
// every: retorna true si TODOS cumplen la condicion
const todosBaratos = productos.every(p => p.precio < 500);
console.log(todosBaratos); // false
Consejo: forEach no retorna nada, mientras que map retorna un nuevo array. Si necesitas transformar datos, usa map. Si solo necesitas ejecutar una accion, usa forEach.

4.3 Ordenamiento y Filtrado

Los metodos de ordenamiento y manipulacion de arrays permiten reorganar, extraer y modificar secciones de un array.

sort y reverse

// sort ordena POR DEFECTO como strings (lexicografico)
const nums = [10, 9, 80, 1, 21];
nums.sort();
console.log(nums); // [1, 10, 21, 80, 9] - INCORRECTO para numeros
// sort con comparador para orden numerico
nums.sort((a, b) => a - b);
console.log(nums); // [1, 9, 10, 21, 80] - ascendente
nums.sort((a, b) => b - a);
console.log(nums); // [80, 21, 10, 9, 1] - descendente
// reverse invierte el orden
const letras = ["a", "b", "c"];
letras.reverse();
console.log(letras); // ["c", "b", "a"]

includes, slice y splice

const arr = ["a", "b", "c", "d", "e"];
// includes: verifica si un elemento existe (retorna boolean)
console.log(arr.includes("c")); // true
console.log(arr.includes("z")); // false
// slice: extrae una porcion SIN modificar el original
const porcion = arr.slice(1, 4); // ["b", "c", "d"]
console.log(arr); // ["a", "b", "c", "d", "e"] - sin cambios
// splice: modifica el array (elimina, reemplaza o agrega)
const original = [1, 2, 3, 4, 5];
// Eliminar 2 elementos desde la posicion 1
const eliminados = original.splice(1, 2);
console.log(eliminados); // [2, 3]
console.log(original);  // [1, 4, 5]
// Insertar elementos en la posicion 1 (sin eliminar)
original.splice(1, 0, "a", "b");
console.log(original); // [1, "a", "b", 4, 5]

✗ Incorrecto

// sort muta el array original
const original = [3, 1, 2];
original.sort((a, b) => a - b);
console.log(original); // [1, 2, 3] - se perdio el orden original

✓ Correcto

// Copia primero para preservar el original
const original = [3, 1, 2];
const ordenado = [...original].sort((a, b) => a - b);
console.log(ordenado);  // [1, 2, 3]
console.log(original);  // [3, 1, 2] - intacto
Cuidado: sort() sin comparador convierte todo a strings y ordena lexicograficamente. Siempre usa un comparador para ordenar numeros.
Consejo: slice() es ideal para crear copias parciales de arrays, mientras que splice() es para modificar el array original.
Ejercicio: Dado un array de objetos con {nombre, calificacion}, ordena de mayor a menor calificacion, extrae los 3 mejores con slice, y verifica si todos tienen calificacion mayor a 70 con every.

4.4 Metodos de Conversion

Estos metodos permiten convertir, aplanar y transformar arrays de diversas formas.

join y flat

// join: convierte un array a string con un separador
const palabras = ["Hola", "mundo", "JS"];
console.log(palabras.join(" "));   // "Hola mundo JS"
console.log(palabras.join("-"));   // "Hola-mundo-JS"
console.log(palabras.join(""));    // "HolamundoJS"
// flat: aplana arrays anidados
const anidado = [1, [2, 3], [4, [5, 6]]];
console.log(anidado.flat());     // [1, 2, 3, 4, [5, 6]]
console.log(anidado.flat(2));    // [1, 2, 3, 4, 5, 6]
console.log(anidado.flat(Infinity)); // [1, 2, 3, 4, 5, 6]

Array.from y Array.of

// Array.from: crea un array desde un iterable o array-like
const texto = "JavaScript";
const letras = Array.from(texto);
console.log(letras); // ["J","a","v","a","S","c","r","i","p","t"]
// Array.from con funcion de mapeo
const nums = Array.from({ length: 5 }, (_, i) => i * 2);
console.log(nums); // [0, 2, 4, 6, 8]
// Array.of: crea un array con los argumentos
const arr1 = Array.of(1, 2, 3);
const arr2 = Array.of(7);
console.log(arr1); // [1, 2, 3]
console.log(arr2); // [7]

entries, keys y values

const frutas = ["manzana", "banana", "naranja"];
// entries: retorna iterador de [indice, valor]
for (const [i, v] of frutas.entries()) {
  console.log(`${i}: ${v}`);
}
// keys: retorna iterador de indices
for (const k of frutas.keys()) {
  console.log(k); // 0, 1, 2
}
// values: retorna iterador de valores
for (const v of frutas.values()) {
  console.log(v); // "manzana", "banana", "naranja"
}
Consejo: flatMap es igual a map() seguido de flat(1). Es util cuando la funcion de mapeo retorna un array y quieres aplanar un nivel.
Ejercicio: Convierte el string "javascript-es-increible" en un array de palabras, capitaliza la primera letra de cada palabra con map, y unelas con join para formar un titulo.

4.5 Creacion y Propiedades de Objetos

Los objetos son colecciones de pares clave-valor. Son la estructura fundamental para modelar entidades en JavaScript.

Literales de Objetos

// Objeto literal
const persona = {
  nombre: "Carlos",
  edad: 28,
  "correo electronico": "carlos@mail.com",
  activo: true
};
// Acceso a propiedades
console.log(persona.nombre);          // "Carlos"
console.log(persona["correo electronico"]); // "carlos@mail.com"
// Agregar propiedades
persona.telefono = "555-1234";
persona["direccion"] = "Calle 5";

Propiedades Dinamicas y Computed Keys

// Propiedades computadas (ES6)
const campo = "edad";
const valor = 30;
const obj = {
  [campo + "Minima"]: 18,
  [`${campo}Maxima`]: 65,
  [campo]: valor
};
console.log(obj); // {edadMinima: 18, edadMaxima: 65, edad: 30}

Enumeracion de Propiedades

const auto = {
  marca: "Toyota",
  modelo: "Corolla",
  anio: 2023
};
// for...in recorre todas las propiedades enumerables
for (let clave in auto) {
  console.log(`${clave}: ${auto[clave]}`);
}
// marca: Toyota
// modelo: Corolla
// anio: 2023
// hasOwnProperty: verifica si la propiedad es propia
console.log(auto.hasOwnProperty("marca"));    // true
console.log(auto.hasOwnProperty("toString")); // false
Cuidado: No confundas punto (.) con corchetes ([]). Usa punto para nombres de propiedades validos, y corchetes cuando el nombre contiene espacios, guiones o es dinamico.
Consejo: En ES6 puedes omitir el valor cuando la clave y la variable tienen el mismo nombre: {nombre} es igual a {nombre: nombre}.
Ejercicio: Crea un objeto "producto" con propiedades nombre, precio y categorias (array). Agrega una propiedad dinamica usando el nombre del producto como clave. Enumera todas las propiedades con for...in.

4.6 Metodos de Objetos

Object ofrece metodos estaticos para trabajar con objetos de forma funcional.

Object.keys, values y entries

const usuario = {
  nombre: "Maria",
  edad: 25,
  email: "maria@mail.com"
};
// Object.keys: array de claves
console.log(Object.keys(usuario));
// ["nombre", "edad", "email"]
// Object.values: array de valores
console.log(Object.values(usuario));
// ["Maria", 25, "maria@mail.com"]
// Object.entries: array de pares [clave, valor]
console.log(Object.entries(usuario));
// [["nombre","Maria"], ["edad",25], ["email","maria@mail.com"]]

Object.assign

// Object.assign: copia propiedades de uno o mas objetos
const defaults = { color: "rojo", tamano: "mediano", activo: true };
const personalizado = { color: "azul", tamano: "grande" };
const resultado = Object.assign({}, defaults, personalizado);
console.log(resultado);
// { color: "azul", tamano: "grande", activo: true }
// Tambien sirve para copiar objetos (shallow copy)
const copia = Object.assign({}, usuario);
copia.nombre = "Pedro";
console.log(usuario.nombre); // "Maria" - original intacto

Object.freeze y Object.seal

// Object.freeze: inmutabilidad total
const congelado = Object.freeze({ x: 1, y: 2 });
congelado.x = 99;        // falla silenciosamente
congelado.z = 3;        // falla silenciosamente
console.log(congelado); // {x: 1, y: 2} - sin cambios
// Object.seal: no permite agregar/eliminar, pero si modificar
const sellado = Object.seal({ a: 1, b: 2 });
sellado.a = 99;        // funciona
sellado.c = 3;        // falla silenciosamente
delete sellado.b;    // falla silenciosamente
console.log(sellado); // {a: 99, b: 2}

✗ Incorrecto

// Copia superficial puede fallar con objetos anidados
const original = { info: { x: 1 } };
const copia = Object.assign({}, original);
copia.info.x = 99;
console.log(original.info.x); // 99 - TAMBIEN CAMBIO!

✓ Correcto

// Usa structuredClone para copia profunda
const original = { info: { x: 1 } };
const copia = structuredClone(original);
copia.info.x = 99;
console.log(original.info.x); // 1 - original intacto
Consejo: Object.keys, values y entries convierten un objeto en algo parecido a un array, lo que te permite usar metodos como map, filter y forEach.
Ejercicio: Dado un objeto de configuracion, crea una funcion que acepte configuraciones por defecto y personalizadas usando Object.assign. Luego congela la configuracion resultante.

4.7 Desestructuracion

La desestructuracion permite extraer valores de arrays y objetos en variables individuales de forma concisa.

Desestructuracion de Arrays

// Basica
const [primero, segundo, tercero] = [10, 20, 30];
console.log(primero); // 10
console.log(tercero); // 30
// Saltar elementos
const [a, , c] = [1, 2, 3];
console.log(a); // 1
console.log(c); // 3
// Rest operator para capturar el resto
const [x, ...resto] = [1, 2, 3, 4, 5];
console.log(x);     // 1
console.log(resto); // [2, 3, 4, 5]

Desestructuracion de Objetos

const usuario = {
  nombre: "Laura",
  edad: 30,
  ciudad: "Madrid",
  pais: "Espana"
};
// Extraer propiedades
const { nombre, edad } = usuario;
console.log(nombre); // "Laura"
// Renombrar variables
const { nombre: userName, ciudad: ubicacion } = usuario;
console.log(userName);  // "Laura"
console.log(ubicacion); // "Madrid"
// Valores por defecto
const { nombre, telefono = "N/A" } = usuario;
console.log(telefono); // "N/A" (no existia en el objeto)

Desestructuracion Anidada

const empresa = {
  nombre: "TechCorp",
  direccion: {
    calle: "Av. Principal",
    numero: 123,
    ciudad: "Lima"
  },
  empleados: ["Ana", "Luis", "Marta"]
};
const {
  nombre: nombreEmpresa,
  direccion: { ciudad, calle },
  empleados: [primerEmpleado]
} = empresa;
console.log(nombreEmpresa);  // "TechCorp"
console.log(ciudad);         // "Lima"
console.log(primerEmpleado); // "Ana"

Desestructuracion en Parametros

function crearUsuario({ nombre, email, rol = "usuario" }) {
  return { nombre, email, rol, activo: true };
}
const user = crearUsuario({
  nombre: "Pedro",
  email: "pedro@mail.com"
});
console.log(user);
// { nombre: "Pedro", email: "pedro@mail.com", rol: "usuario", activo: true }
Cuidado: Si desestructuras una propiedad que no existe y no proporcionas un valor por defecto, la variable sera undefined.
Consejo: Cuando renombres propiedades, el nombre de la variable es el nuevo nombre, no el original. const { nombre: x } = obj crea una variable "x", no "nombre".
Ejercicio: Crea una funcion que reciba un objeto anidado {perfil: {nombre, edad}, contactos: {email, telefono}} y use desestructuracion anidada para extraer todos los campos con valores por defecto.

4.8 Spread y Rest Operator

El operador spread (...) expansiona elementos, mientras que el operador rest (...) agrupa elementos. Ambos usan la misma sintaxis pero con propositos opuestos.

Spread en Arrays

// Copiar un array
const original = [1, 2, 3];
const copia = [...original];
copia.push(4);
console.log(original); // [1, 2, 3] - intacto
console.log(copia);     // [1, 2, 3, 4]
// Combinar arrays
const a = [1, 2];
const b = [3, 4];
const combinado = [...a, ...b, 5];
console.log(combinado); // [1, 2, 3, 4, 5]

Spread en Objetos

// Copiar y extender objetos
const config = { tema: "oscuro", idioma: "es" };
const configUser = { ...config, usuario: "admin", tema: "claro" };
console.log(configUser);
// { tema: "claro", idioma: "es", usuario: "admin" }

Rest en Parametros

// Rest param: captura argumentos sobrantes
function sumar(...numeros) {
  return numeros.reduce((acc, n) => acc + n, 0);
}
console.log(sumar(1, 2, 3));         // 6
console.log(sumar(10, 20, 30, 40)); // 100
// Combina con parametros normales
function log(nivel, ...mensajes) {
  console.log(`[${nivel}]`, ...mensajes);
}
log("INFO", "Servidor", "iniciado", "puerto 3000");
// [INFO] Servidor iniciado puerto 3000

✗ Incorrecto

// concat crea un nuevo array innecesariamente
const arr1 = [1, 2];
const arr2 = [3, 4];
const merged = arr1.concat(arr2);

✓ Correcto

// Spread es mas legible y flexible
const arr1 = [1, 2];
const arr2 = [3, 4];
const merged = [...arr1, ...arr2, 5];
Cuidado: El spread solo crea una copia superficial (shallow). Para objetos con propiedades anidadas, usa structuredClone() para una copia profunda.
Consejo: El spread para objetos es ideal para simular actualizaciones parciales en inmutabilidad, como en Redux o React state.
Ejercicio: Crea dos objetos de configuracion, combinalos con spread donde el segundo sobreescribe las propiedades compartidas. Luego crea una funcion que acepte parametros normales mas rest y la invoques con diferentes argumentos.

4.9 JSON

JSON (JavaScript Object Notation) es el formato estandar para intercambiar datos entre servidores y clientes.

JSON.stringify

// Convierte un objeto a string JSON
const usuario = {
  nombre: "Carlos",
  edad: 28,
  hobbies: ["leer", "programar"],
  activo: true
};
const jsonStr = JSON.stringify(usuario);
console.log(jsonStr);
// Con formato legible (pretty print)
console.log(JSON.stringify(usuario, null, 2));
// Serializar con filtro usando un array de claves
console.log(JSON.stringify(usuario, ["nombre", "edad"]));
// '{"nombre":"Carlos","edad":28}'

JSON.parse

// Convierte un string JSON a objeto
const jsonStr = '{"nombre":"Ana","edad":25,"activo":true}';
const obj = JSON.parse(jsonStr);
console.log(obj.nombre); // "Ana"
// Con reviver para transformar valores durante el parseo
const jsonFecha = '{"fecha":"2024-01-15"}';
const conFecha = JSON.parse(jsonFecha, (clave, valor) => {
  if (clave === "fecha") return new Date(valor);
  return valor;
});
console.log(conFecha.fecha); // Date object

Copia Profunda con JSON

// Copia profunda clasica (pre-ES2022)
const original = {
  datos: { x: 1, y: 2 },
  lista: [1, 2, 3]
};
const copia = JSON.parse(JSON.stringify(original));
copia.datos.x = 99;
console.log(original.datos.x); // 1 - original intacto
// Moderno: structuredClone (recomendado)
const copia2 = structuredClone(original);

Validacion de JSON

// Funcion para validar si un string es JSON valido
function esJSONValido(str) {
  try {
    JSON.parse(str);
    return true;
  } catch (e) {
    return false;
  }
}
console.log(esJSONValido('{"a":1}'));  // true
console.log(esJSONValido('{a: 1}'));   // false
console.log(esJSONValido('not json')); // false
Cuidado: JSON.stringify no puede serializar funciones, undefined, Symbol, ni circular references. Estos valores se omiten o se convierten en null.
Consejo: El patron JSON.parse(JSON.stringify(obj)) es una forma rapida de hacer copia profunda, pero structuredClone es mas moderno y maneja mas tipos de datos.
Ejercicio: Crea un objeto con datos anidados, serializalo con JSON.stringify, modifica el original, y deserializa con JSON.parse. Verifica que la copia se mantiene intacta.

5.1 Introduccion al DOM

El DOM (Document Object Model) es una representacion en memoria de la estructura HTML de una pagina. JavaScript puede leer y modificar esta representacion para cambiar lo que el usuario ve.

El Arbol de Nodos

El navegador convierte el HTML en un arbol de nodos donde cada elemento HTML es un nodo. Cada nodo tiene relaciones padre-hijo-hermano.

// El objeto window es el objeto global en el navegador
console.log(window);          // objeto Window
console.log(window.document); // referencia al DOM
console.log(document);         // shorthand de window.document
// Propiedades basicas del documento
console.log(document.title);   // titulo de la pagina
console.log(document.body);    // nodo body
console.log(document.head);    // nodo head
console.log(document.documentElement); // nodo html raiz

Tipos de Nodos

const elemento = document.querySelector("h1");
console.log(elemento.nodeType);  // 1 (ELEMENT_NODE)
console.log(elemento.nodeName); // "H1"
console.log(elemento.nodeValue); // null
// Tipos comunes de nodos:
// 1 - Element (div, p, h1, etc.)
// 3 - Text (texto dentro de un elemento)
// 8 - Comment (<!-- comentario -->)
// 9 - Document (el nodo raiz)

Navegacion entre Nodos

const lista = document.querySelector("ul");
const primerItem = lista.firstElementChild;
const ultimoItem = lista.lastElementChild;
// Navegar hacia arriba
const padre = primerItem.parentElement;
// Navegar entre hermanos
const siguiente = primerItem.nextElementSibling;
const anterior = ultimoItem.previousElementSibling;
// Todos los hijos
console.log(lista.children);    // HTMLCollection de elementos
console.log(lista.childNodes);  // NodeList de todos los nodos
Consejo: Prefiere children sobre childNodes cuando solo te interesan los elementos HTML y no los nodos de texto o comentarios.
Ejercicio: Abre la consola del navegador en cualquier pagina web y usa document.body para explorar la estructura. Navega desde body hasta encontrar el titulo de la pagina usando firstElementChild y lastElementChild.

5.2 Selectores del DOM

Para manipular elementos primero necesitas encontrarlos. JavaScript ofrece varios metodos para seleccionar elementos del DOM.

getElementById

// Selecciona UN elemento por su atributo id
const titulo = document.getElementById("titulo");
console.log(titulo); // elemento <h1 id="titulo">
// Retorna null si no encuentra el elemento
const noExiste = document.getElementById("no-existe");
console.log(noExiste); // null

querySelector y querySelectorAll

// querySelector: retorna el PRIMER elemento que coincida
const primBtn = document.querySelector("button");
const activo = document.querySelector(".activo");
const enlace = document.querySelector("nav a[href='#inicio']");
// querySelectorAll: retorna NodeList con TODOS los elementos
const todosBotones = document.querySelectorAll("button");
const items = document.querySelectorAll(".lista-item");
// Iterar sobre NodeList
items.forEach(item => {
  console.log(item.textContent);
});
// Convertir a array para usar mas metodos
const itemsArray = Array.from(items);
const filtrados = itemsArray.filter(el => el.classList.contains("activo"));

getElementsByClassName y getElementsByTagName

// Retorna HTMLCollection (live - se actualiza automaticamente)
const cards = document.getElementsByClassName("card");
console.log(cards.length);
// HTMLCollection no tiene forEach, usa for...of o Array.from
for (const card of cards) {
  console.log(card.textContent);
}
// Por etiqueta
const parrafos = document.getElementsByTagName("p");

✗ Incorrecto

// getElementsByClassName retorna HTMLCollection live
const items = document.getElementsByClassName("item");
// No tiene .filter(), .map(), etc.
items.filter(...); // ERROR: filter is not a function

✓ Correcto

// querySelectorAll retorna NodeList estatica
const items = document.querySelectorAll(".item");
// Tiene .forEach(), y puedes usar Array.from para mas
Array.from(items).filter(...); // funciona
Consejo: querySelector y querySelectorAll son los mas versatiles porque aceptan cualquier selector CSS valido. Son la recomendacion para la mayoria de casos.

5.3 Modificacion de Elementos

Una vez que tienes un elemento seleccionado, puedes modificar su contenido, estilos y atributos.

Contenido de Texto

const parrafo = document.querySelector("p");
// textContent: obtiene/establece todo el texto (incluye oculto)
parrafo.textContent = "Nuevo texto del parrafo";
// innerText: similar pero respeta estilos CSS
parrafo.innerText = "Texto visible";
// innerHTML: obtiene/establece HTML interno (usar con cuidado)
parrafo.innerHTML = "<strong>Texto en negrita</strong>";
// innerHTML con template literals para contenido dinamico
const nombre = "Carlos";
parrafo.innerHTML = `Hola <strong>${nombre}</strong>, bienvenido!`;

Manipulacion de Clases con classList

const elemento = document.querySelector(".card");
// Agregar clase
elemento.classList.add("activo");
elemento.classList.add("visible", "destacado");
// Eliminar clase
elemento.classList.remove("visible");
// Alternar clase (toggle)
elemento.classList.toggle("activo");
// Verificar si tiene una clase
if (elemento.classList.contains("activo")) {
  console.log("Elemento esta activo");
}
// Reemplazar una clase
elemento.classList.replace("viejo", "nuevo");

Atributos

const link = document.querySelector("a");
// Establecer atributos
link.setAttribute("href", "https://ejemplo.com");
link.setAttribute("target", "_blank");
// Obtener atributos
console.log(link.getAttribute("href"));
// Eliminar atributos
link.removeAttribute("target");
// Verificar si tiene un atributo

Cuidado: Nunca uses innerHTML con contenido que introduzca el usuario. Esto puede provocar ataques de XSS (Cross-Site Scripting). Usa textContent para texto plano.
Consejo: toggle es ideal para menus y modales: si esta abierto lo cierra, si esta cerrado lo abre, todo en una sola linea.

5.4 Creacion y Eliminacion de Elementos

JavaScript puede crear, insertar y eliminar elementos HTML dinamicamente, permitiendo interfaces interactivas.

Crear y Agregar Elementos

// Crear un elemento nuevo
const nuevoParrafo = document.createElement("p");
nuevoParrafo.textContent = "Soy un parrafo nuevo!";
nuevoParrafo.classList.add("destacado");
// appendChild: agrega al final del padre
const contenedor = document.querySelector(".lista");
contenedor.appendChild(nuevoParrafo);
// insertBefore: agrega antes de un hermano especifico
const referencia = contenedor.firstElementChild;
contenedor.insertBefore(nuevoParrafo, referencia);
// insertAdjacentHTML: inserta HTML en una posicion relativa
// beforebegin, afterbegin, beforeend, afterend
contenedor.insertAdjacentHTML("afterbegin", "<li>Primer item</li>");

Eliminar y Reemplazar Elementos

// removeChild: elimina un hijo especifico
const hijo = contenedor.querySelector("li:first-child");
contenedor.removeChild(hijo);
// Moderno: directo remove
hijo.remove();
// replaceChild: reemplaza un hijo con otro
const viejo = contenedor.querySelector("p");
const nuevo = document.createElement("h2");
nuevo.textContent = "Titulo reemplazado";
contenedor.replaceChild(nuevo, viejo);

DocumentFragment para Rendimiento

// Agregar multiples elementos de forma eficiente
const fragment = document.createDocumentFragment();
for (let i = 0; i < 100; i++) {
  const li = document.createElement("li");
  li.textContent = `Item ${i}`;
  fragment.appendChild(li);
}
// Una sola operacion DOM (mucho mas rapido)
document.querySelector("ul").appendChild(fragment);
Cuidado: Agregar elementos uno por uno con appendChild en un bucle es lento porque re-renderiza el DOM en cada iteracion. Usa DocumentFragment para agregar muchos elementos de golpe.
Consejo: insertAdjacentHTML es muy practico para agregar HTML complejo sin crear cada elemento manualmente con createElement.

5.5 Eventos del DOM

Los eventos permiten que JavaScript responda a acciones del usuario como clics, teclas, envio de formularios, y mas.

addEventListener

const boton = document.querySelector("#miBoton");
// Escuchar un evento
boton.addEventListener("click", function(evento) {
  console.log("Boton clickeado!");
  console.log("Tipo:", evento.type);
  console.log("Target:", evento.target);
});
// Con arrow function (mas comun)
boton.addEventListener("click", (e) => {
  console.log("Clic!", e.target);
});
// Con opciones (once, passive, capture)
boton.addEventListener("click", handler, { once: true });
// Se ejecuta solo una vez y se remueve automaticamente

removeEventListener

// Para remover, necesitas la misma referencia de funcion
function manejarClick(e) {
  console.log("Click!");
}
boton.addEventListener("click", manejarClick);
// Ahora si puedo removerlo
boton.removeEventListener("click", manejarClick);
// NO funciona con arrow functions anonimas:
boton.addEventListener("click", (e) => console.log("hi"));
boton.removeEventListener("click", (e) => console.log("hi"));
// NO funciona - son diferentes referencias

El Objeto Event

boton.addEventListener("click", (e) => {
  console.log(e.type);       // "click"
  console.log(e.target);      // el elemento clickeado
  console.log(e.currentTarget); // el elemento con el listener
  // preventDefault: evita el comportamiento por defecto
  e.preventDefault();
  // stopPropagation: detiene la propagacion del evento
  e.stopPropagation();
});
Cuidado: No uses onclick = funcion como addEventListener. onclick sobreescribe handlers anteriores, mientras que addEventListener permite multiples listeners.
Consejo: Usa { once: true } como tercera opcion de addEventListener cuando el listener solo debe ejecutarse una vez.
Ejercicio: Crea un boton que registre cuantas veces se ha clickeado. Tambien agrega un listener al link que prevenga la navegacion y muestre un mensaje antes de redirigir.

5.6 Eventos Comunes

El DOM ofrece muchos tipos de eventos para diferentes interacciones del usuario. Conocer los mas comunes es esencial.

6.1 Introduccion a la Asincronia

JavaScript es single-threaded (un solo hilo), pero puede manejar operaciones asincronas como peticiones HTTP, temporizadores y eventos del usuario sin bloquear la ejecucion.

Sincrono vs Asincrono

// SINCRONO: se ejecuta linea por linea, bloquea
console.log("Uno");
console.log("Dos");
console.log("Tres");
// Salida: Uno, Dos, Tres (en ese orden)
// ASINCRONO: se agenda para despues
console.log("Inicio");
setTimeout(() => console.log("Asincrono"), 0);
console.log("Fin");
// Salida: Inicio, Fin, Asincrono

Call Stack

El Call Stack es una pila LIFO (Last In, First Out) que registra las funciones que se estan ejecutando.

function a() { b(); }
function b() { c(); }
function c() { console.log("fin"); }
a();
// Call Stack:
// 1. a() entra
// 2. b() entra (dentro de a)
// 3. c() entra (dentro de b)
// 4. c() sale - imprime "fin"
// 5. b() sale
// 6. a() sale

Event Loop

El Event Loop verifica continuamente si el Call Stack esta vacio y si hay tareas pendientes.

console.log("1: Sincrono");
setTimeout(() => console.log("2: Timeout"), 0);
Promise.resolve().then(() => console.log("3: Microtask"));
console.log("4: Sincrono");
// Salida:
// 1: Sincrono    (Call Stack)
// 4: Sincrono    (Call Stack)
// 3: Microtask   (Microtask Queue - prioridad)
// 2: Timeout     (Callback Queue)
Consejo: Las Microtasks (Promesas) tienen prioridad sobre las Callback Queue (setTimeout, eventos). Esto es importante para predecir el orden de ejecucion.

6.3 Promesas

Una Promesa es un objeto que representa el resultado eventual de una operacion asincrona. Puede estar en tres estados: pending, fulfilled o rejected.

Crear una Promesa

const miPromesa = new Promise((resolve, reject) => {
  const exito = true;
  setTimeout(() => {
    if (exito) {
      resolve("Operacion exitosa!");
    } else {
      reject(new Error("Algo fallo"));
    }
  }, 1000);
});

then, catch y finally

miPromesa
  .then((resultado) => {
    console.log(resultado); // "Operacion exitosa!"
  })
  .catch((error) => {
    console.error(error.message); // "Algo fallo"
  })
  .finally(() => {
    console.log("Terminado (exito o error)");
  });

Encadenamiento de Promesas

function obtenerUsuario(id) {
  return new Promise((resolve) => {
    setTimeout(() => {
      resolve({ id, nombre: "Carlos", rolId: 3 });
    }, 500);
  });
}
function obtenerRol(rolId) {
  return new Promise((resolve) => {
    setTimeout(() => {
      resolve({ id: rolId, nombre: "Admin" });
    }, 300);
  });
}
// Encadenar promesas
obtenerUsuario(1)
  .then(usuario => {
    console.log(usuario.nombre);
    return obtenerRol(usuario.rolId);
  })
  .then(rol => {
    console.log(rol.nombre); // "Admin"
  })
  .catch(err => console.error(err));
Consejo: Cada .then() retorna una nueva promesa, lo que permite encadenarlas. Si un .then() retorna un valor, el siguiente .then() lo recibe. Si retorna una promesa, el siguiente espera a que se resuelva.

6.4 Metodos Estaticos de Promesas

Promise ofrece metodos estaticos para trabajar con multiples promesas de forma conjunta.

Promise.all

// Promise.all: espera que TODAS se resuelvan
// Si una falla, todas fallan
const p1 = fetch("/api/usuarios").then(r => r.json());
const p2 = fetch("/api/productos").then(r => r.json());
const p3 = fetch("/api/pedidos").then(r => r.json());
Promise.all([p1, p2, p3])
  .then(([usuarios, productos, pedidos]) => {
    console.log(usuarios, productos, pedidos);
  })
  .catch((error) => {
    console.error("Una fallo:", error);
  });

Promise.allSettled

// Promise.allSettled: espera que todas terminen (resueltas o rechazadas)
// Nunca rechaza - retorna el estado de cada una
const resultados = await Promise.allSettled([
  Promise.resolve("exito 1"),
  Promise.reject(new Error("fallo")),
  Promise.resolve("exito 2")
]);
// [{status:"fulfilled", value:"exito 1"},
//  {status:"rejected", reason: Error},
//  {status:"fulfilled", value:"exito 2"}]
resultados.forEach(r => {
  if (r.status === "fulfilled") {
    console.log("OK:", r.value);
  } else {
    console.log("Fallo:", r.reason);
  }
});

Promise.race y Promise.any

// Promise.race: resuelve/rechaza con la PRIMERA que termine
const rapida = new Promise(r => setTimeout(() => r("rapida"), 100));
const lenta = new Promise(r => setTimeout(() => r("lenta"), 2000));
const resultado = await Promise.race([rapida, lenta]);
console.log(resultado); // "rapida"
// Promise.any: resuelve con la PRIMERA EXITOSA
// Si todas fallan, rechaza con AggregateError
const a = Promise.reject("fallo a");
const b = Promise.resolve("exito b");
const c = Promise.resolve("exito c");
const r = await Promise.any([a, b, c]);
console.log(r); // "exito b" (la primera exitosa)

Promise.resolve y Promise.reject

// Promise.resolve: crea una promesa ya resuelta
const resuelta = Promise.resolve(42);
resuelta.then(v => console.log(v)); // 42
// Promise.reject: crea una promesa ya rechazada
const rechazada = Promise.reject(new Error("fallo"));
rechazada.catch(e => console.error(e.message)); // "fallo"

✗ Incorrecto

// Promise.all falla si una sola promesa rechaza
try {
  const [a, b] = await Promise.all([
    fetch("/api/a"),
    fetch("/api/b") // si esta falla, todo falla
  ]);
} catch (e) { // pierdes ambos resultados
}

✓ Correcto

// allSettled es mas resiliente
const [a, b] = await Promise.allSettled([
  fetch("/api/a"),
  fetch("/api/b")
]);
// Procesas cada resultado individualmente
Consejo: Usa Promise.allSettled cuando necesitas el resultado de todas las operaciones aunque algunas fallen. Usa Promise.all cuando todas son obligatorias.

6.5 Async/Await

async/await es azucar sintactico sobre las promesas que permite escribir codigo asincrono que se ve y comporta como sincrono.

Sintaxis basica

// Funcion asincrona
async function obtenerDatos() {
  const respuesta = await fetch("https://api.ejemplo.com/datos");
  const datos = await respuesta.json();
  return datos;
}
// async function SIEMPRE retorna una promesa
obtenerDatos().then(d => console.log(d));

Manejo de Errores con try/catch

async function cargarUsuario(id) {
  try {
    const res = await fetch(`/api/usuarios/${id}`);
    if (!res.ok) {
      throw new Error(`HTTP ${res.status}: ${res.statusText}`);
    }
    const usuario = await res.json();
    return usuario;
  } catch (error) {
    console.error("Error al cargar:", error.message);
    return null;
  } finally {
    console.log("Peticion completada");
  }
}

Secuencial vs Paralelo

// SECUENCIAL: una despues de otra (lento)
async function secuencial() {
  const usuarios = await fetchUsuarios();   // 2s
  const productos = await fetchProductos(); // 2s
  return { usuarios, productos };
  // Total: 4 segundos
}
// PARALELO: todas al mismo tiempo (rapido)
async function paralelo() {
  const [usuarios, productos] = await Promise.all([
    fetchUsuarios(),
    fetchProductos()
  ]);
  return { usuarios, productos };
  // Total: 2 segundos
}
Cuidado: await dentro de un bucle for ejecuta secuencialmente. Si quieres paralelismo, usa Promise.all con map().

6.7 Manejo de Errores en Asincronia

Un manejo robusto de errores es critico en codigo asincrono. Existen diferentes patrones para capturar y recuperarse de errores.

try/catch en async/await

async function cargarDatos() {
  try {
    const res = await fetch("/api/datos");
    if (!res.ok) throw new Error(`HTTP ${res.status}`);
    return await res.json();
  } catch (error) {
    console.error("Error cargando datos:", error.message);
    return []; // valor por defecto
  }
}

.catch en Promesas

// Siempre ten un .catch al final de la cadena
fetch("/api/datos")
  .then(r => r.json())
  .then(datos => procesar(datos))
  .catch(error => {
    console.error("Fallo:", error);
    mostrarError("No se pudieron cargar los datos");
  });

Patron de Reintento

async function fetchConReintentos(url, maxReintentos = 3) {
  for (let i = 0; i < maxReintentos; i++) {
    try {
      const res = await fetch(url);
      if (res.ok) return await res.json();
      if (res.status < 500) throw new Error(`Client error: ${res.status}`);
    } catch (error) {
      if (i === maxReintentos - 1) throw error;
      const espera = Math.pow(2, i) * 1000; // backoff exponencial
      console.log(`Reintento ${i + 1} en ${espera}ms...`);
      await new Promise(r => setTimeout(r, espera));
    }
  }
}

Timeout con Promise

function conTimeout(promesa, milisegundos) {
  const timeout = new Promise((_, reject) => {
    setTimeout(() => reject(
      new Error(`Timeout despues de ${milisegundos}ms`)
    ), milisegundos);
  });
  return Promise.race([promesa, timeout]);
}
// Uso
try {
  const datos = await conTimeout(fetch("/api/lento"), 5000);
} catch (e) {
  console.error(e.message); // "Timeout despues de 5000ms"
}
Cuidado: Un error no atrapado en una promesa genera un Unhandled Promise Rejection. Siempre usa .catch() o try/catch con async/await.
Consejo: Usa backoff exponencial (espera duplicada cada reintento) para no saturar el servidor con reintentos rapidos.

6.8 Patrones Asincronos Avanzados

Patrones avanzados para controlar la ejecucion asincrona: paralelo, secuencial, throttling, debouncing y cancelacion.

Ejecucion Paralela

// Ejecutar multiples tareas simultaneamente
async function cargarTodo() {
  const [usuarios, productos, categorias] = await Promise.all([
    fetch("/api/usuarios").then(r => r.json()),
    fetch("/api/productos").then(r => r.json()),
    fetch("/api/categorias").then(r => r.json())
  ]);

// Parallel con limite de concurrencia
async function parallelLimit(tareas, limite) {
  const resultados = [];
  const ejecutando = new Set();
  for (const tarea of tareas) {
    const p = tarea().then(r => {
      ejecutando.delete(p);
      return r;
    });
    ejecutando.add(p);
    resultados.push(p);
    if (ejecutando.size >= limite) {
      await Promise.race(ejecutando);
    }
  }
  return Promise.all(resultados);
}

Ejecucion Secuencial

// Ejecutar tareas una por una en orden
async function ejecutarSecuencialmente(tareas) {
  const resultados = [];
  for (const tarea of tareas) {
    const resultado = await tarea();
    resultados.push(resultado);
  }
  return resultados;
}
const tareas = [
  () => fetch("/api/paso1").then(r => r.json()),
  (prev) => fetch(`/api/paso2/${prev.id}`).then(r => r.json()),
  (prev) => fetch(`/api/paso3/${prev.id}`).then(r => r.json())
];

Debouncing y Throttling

// DEBOUNCE: ejecutar despues de una pausa
function debounce(fn, delay) {
  let timerId;
  return function(...args) {
    clearTimeout(timerId);
    timerId = setTimeout(() => fn(...args), delay);
  };
}
// Ejemplo: busqueda en tiempo real
const buscar = debounce((termino) => {
  fetch(`/api/buscar?q=${termino}`);
}, 300);
// THROTTLE: ejecutar como maximo una vez por periodo
function throttle(fn, limite) {
  let ultimaVez = 0;
  return function(...args) {
    const ahora = Date.now();
    if (ahora - ultimaVez >= limite) {
      ultimaVez = ahora;
      fn(...args);
    }
  };
}
// Ejemplo: scroll que se ejecuta max 1 vez por 200ms
addEventListener("scroll", throttle(() => {
  console.log("Scroll:", window.scrollY);
}, 200));

AbortController

// Cancelar peticiones fetch
const controller = new AbortController();
fetch("/api/lento", { signal: controller.signal })
  .then(r => r.json())
  .catch(e => {
    if (e.name === "AbortError") {
      console.log("Peticion cancelada");
    }
  });
// Cancelar despues de 5 segundos
setTimeout(() => controller.abort(), 5000);
Debounce vs Throttle: debounce retrasa la ejecucion hasta que haya silencio (busqueda en input). throttle limita la frecuencia (scroll, resize). debounce espera, throttle ejecuta inmediatamente y luego espera.

7.1 Introduccion a POO

La Programacion Orientada a Objetos (POO) es un paradigma que organiza el codigo en torno a objetos, que agrupan datos (propiedades) y comportamiento (metodos). JavaScript implementa POO basada en prototipos, lo que la hace unica.

Objetos Literales

La forma mas simple de crear objetos en JavaScript es usando la notacion literal con llaves:

// Objeto literal basico
const persona = {
  nombre: "Carlos",
  edad: 28,
  activo: true,
  saludar() {
    return "Hola, soy " + this.nombre;
  }
};

Notacion de Punto y Corchete

Existen dos formas de acceder a las propiedades de un objeto:

const usuario = { nombre: "Ana", edad: 25 };

// Notacion de punto
console.log(usuario.nombre); // "Ana"

// Notacion de corchete (dinamica)
const campo = "edad";
console.log(usuario[campo]); // 25

// El corchete permite nombres dinamicos
const key = "email";
usuario[key] = "ana@email.com";
Tip: Usa la notacion de corchete cuando el nombre de la propiedad se determina en tiempo de ejecucion. Para propiedades conocidas, la notacion de punto es mas legible.

Destructuracion de Objetos

ES6 introduce una forma elegante de extraer propiedades de objetos:

const animal = { tipo: "perro", nombre: "Max", edad: 5 };

// Destructuracion basica
const { tipo, nombre } = animal;
console.log(tipo, nombre); // "perro" "Max"

// Renombrar propiedades
const { nombre: nombreAnimal, edad: edadAnimal } = animal;

// Valor por defecto
const { raza = "desconocida" } = animal;
console.log(raza); // "desconocida"
Cuidado: this en metodos de objetos literales apunta al propio objeto, pero puede cambiar si pasas el metodo como callback. Usa arrow functions con cuidado dentro de objetos.

7.2 Funciones Constructoras

Las funciones constructoras son el mecanismo tradicional de JavaScript para crear multiples instancias de un objeto con la misma estructura. Se usan con la palabra clave new.

Constructor Basico y new

function Persona(nombre, edad) {
  this.nombre = nombre;
  this.edad = edad;
}

Persona.prototype.saludar = function() {
  return "Hola, soy " + this.nombre + " con " + this.edad + " anios";
};

const p1 = new Persona("Luis", 30);
const p2 = new Persona("Maria", 25);
console.log(p1.saludar()); // "Hola, soy Luis con 30 anios"

Prototype

Cada funcion tiene un objeto prototype que comparten todas las instancias. Esto es mas eficiente que definir metodos dentro del constructor:

function Vehiculo(marca, velocidad) {
  this.marca = marca;
  this.velocidad = velocidad;
}

Vehiculo.prototype.acelerar = function(incremento) {
  this.velocidad += incremento;
  return this.velocidad;
};

Vehiculo.prototype.frenar = function() {
  this.velocidad = 0;
};

const carrito = new Vehiculo("Toyota", 0);
carrito.acelerar(60); // 60
carrito.frenar();      // 0
Tip: Define metodos en el prototype, no dentro del constructor. Esto evita que cada instancia cree una nueva funcion en memoria, ahorrando recursos significativamente.

Patron de Objetos con Prototipos

Puedes encadenar constructores usando call o apply para reutilizar logica:

function Animal(nombre) {
  this.nombre = nombre;
}

function Perro(nombre, raza) {
  Animal.call(this, nombre);
  this.raza = raza;
}

// Heredar prototype
Perro.prototype = Object.create(Animal.prototype);
Perro.prototype.constructor = Perro;

Perro.prototype.ladrar = function() {
  return "Guau!";
};

const rex = new Perro("Rex", "Pastor Aleman");
console.log(rex.ladrar()); // "Guau!"
Cuidado: No olvides asignar Perro.prototype.constructor = Perro despues de sobrescribir el prototype. Sin esto, instanceof y constructor pueden comportarse de forma inesperada.

ES6 introdujo la sintaxis de clases en JavaScript, que ofrece una forma mas clara y organizada de crear constructores y manejar herencia, aunque internamente sigue siendo basada en prototipos.

Clase Basica

    this.email = email;
    this.activo = true;

7.4 Herencia y Polimorfismo

La herencia permite crear clases que reutilizan y extienden el comportamiento de otras. El polimorfismo permite que objetos de diferentes clases respondan a la misma interfaz de forma diferente.

Herencia con extends y super

class Animal {
  constructor(nombre, tipo) {
    this.nombre = nombre;
    this.tipo = tipo;
  }

  describir() {
    return this.nombre + " es un " + this.tipo;
  }

  hacerSonido() {
    return "Sonido generico";
  }
}

class Perro extends Animal {
  constructor(nombre, raza) {
    super(nombre, "perro");
    this.raza = raza;
  }

  hacerSonido() {
    return "Guau guau!";
  }

  buscar() {
    return this.nombre + " esta buscando la pelota";
  }
}

const rocky = new Perro("Rocky", "Labrador");
console.log(rocky.describir());  // Heredado de Animal
console.log(rocky.hacerSonido()); // "Guau guau!"
console.log(rocky.buscar());

Polimorfismo en Accion

class Gato extends Animal {
  constructor(nombre) {
    super(nombre, "gato");
  }

  hacerSonido() {
    return "Miau!";
  }
}

// Polimorfismo: misma interfaz, diferentes comportamientos
function hacerRuido(animal) {
  console.log(animal.nombre + " dice: " + animal.hacerSonido());
}

const mascotas = [
  new Perro("Max", "Pastor"),
  new Gato("Luna")

mascotas.forEach(hacerRuido);
// "Max dice: Guau guau!"
// "Luna dice: Miau!"

Verificacion con instanceof

console.log(rocky instanceof Perro);  // true
console.log(rocky instanceof Animal); // true
console.log(rocky instanceof Gato);   // false

✗ Incorrecto

// Super debe ser llamado ANTES de this
class Malo extends Animal {
  constructor(nombre) {
    this.nuevo = "x"; // ReferenceError!
    super(nombre);
  }
}

✓ Correcto

// Super primero, this despues
class Bueno extends Animal {
  constructor(nombre) {
    super(nombre);
    this.nuevo = "x"; // OK!
  }
}
Cuidado: Siempre llama a super() antes de usar this en un constructor de clase hija. Si no lo haces, JavaScript lanzara un ReferenceError.

7.5 Encapsulamiento

El encapsulamiento protege el estado interno de un objeto, exponiendo solo lo necesario. JavaScript ofrece varias estrategias para lograrlo, desde convenciones hasta campos privados nativos.

Closures como Privado

function CrearCuenta(titular, saldoInicial) {
  let saldo = saldoInicial; // Variable privada via closure

  return {
    depositar(monto) {
      if (monto > 0) saldo += monto;
    },
    extraer(monto) {
      if (monto > 0 && monto <= saldo) saldo -= monto;
    },
    verSaldo() {
      return saldo;
    }
  };
}

const cuenta = CrearCuenta("Laura", 1000);

console.log(cuenta.saldo);         // undefined (privado)

    return datos.password === intento;
  }

  getNombre() {
    return datosPrivados.get(this).nombre;
  }
}

    if (pin !== this.#pin) return "PIN incorrecto";
    if (monto > this.#saldo) return "Saldo insuficiente";
    this.#saldo -= monto;
    return "Retiro exitoso. Saldo: " + this.#saldo;

7.6 Patrones de Diseno

Los patrones de diseno son soluciones reutilizables a problemas comunes en el desarrollo de software. JavaScript, con su flexibilidad, permite implementar multiples patrones de forma natural.

Singleton

Asegura que solo exista una instancia de una clase en toda la aplicacion:

class Configuracion {
  static #instancia = null;

  static obtenerInstancia() {
    if (!Configuracion.#instancia) {
      Configuracion.#instancia = new Configuracion();
    }
    return Configuracion.#instancia;
  }

  constructor() {
    if (Configuracion.#instancia) {
      throw new Error("Usa Configuracion.obtenerInstancia()");
    }
    this.tema = "oscuro";
    this.idioma = "es";
  }
}

const c1 = Configuracion.obtenerInstancia();
const c2 = Configuracion.obtenerInstancia();
console.log(c1 === c2); // true - misma instancia

Factory

Crea objetos sin especificar la clase exacta a instanciar:

class NotificacionEmail {
  enviar(msg) { return "Email: " + msg; }
}

class NotificacionSMS {
  enviar(msg) { return "SMS: " + msg; }
}

class NotificacionPush {
  enviar(msg) { return "Push: " + msg; }
}

function CrearNotificacion(tipo) {
  const map = {
    email: NotificacionEmail,
    sms: NotificacionSMS,
    push: NotificacionPush
  };
  return new map[tipo]();
}

console.log(n.enviar("Hola!")); // "SMS: Hola!"

Observer

Permite que un objeto notifique a multiples observadores cuando cambia su estado:

class EventEmitter {
  #listeners = {};

  on(evento, callback) {
    if (!this.#listeners[evento]) {
      this.#listeners[evento] = [];
    }
    this.#listeners[evento].push(callback);
  }

  emit(evento, ...args) {
    if (this.#listeners[evento]) {
      this.#listeners[evento].forEach(cb => cb(...args));
    }
  }
}

const emitter = new EventEmitter();
emitter.on("mensaje", (texto) => console.log("Receptor 1:", texto));
emitter.on("mensaje", (texto) => console.log("Receptor 2:", texto));

Tip: El patron Observer es la base de los eventos en JavaScript (addEventListener) y de frameworks reactivos como Vue y React. Entenderlo te ayuda a comprender como funcionan.
Cuidado: En el patron Observer, asegurate de implementar un metodo para cancelar suscripciones. De lo contrario, podrias tener fugas de memoria cuando los objetos observados sigan notificando a observadores que ya no necesitan las notificaciones.

7.7 Prototipos y Cadenas de Prototipos

Los prototipos son el mecanismo fundamental de herencia en JavaScript. Cada objeto tiene un enlace a otro objeto llamado su prototipo, formando una cadena que termina en null.

El Objeto prototype

function Mascota(nombre) {
  this.nombre = nombre;
}

// Prototype es una propiedad de la funcion
Mascota.prototype.saludar = function() {
  return "Hola soy " + this.nombre;
};

const gato = new Mascota("Michi");

// La cadena: gato - Mascota.prototype - Object.prototype - null
console.log(gato.__proto__ === Mascota.prototype); // true
console.log(Mascota.prototype.__proto__ === Object.prototype); // true

Object.create()

Crea un nuevo objeto con un prototipo especifico, permitiendo herencia manual:

const prototipoAnimal = {
  tipo: "desconocido",
  describir() {
    return "Soy un " + this.tipo + " llamado " + this.nombre;
  }
};

perro.nombre = "Toby";
perro.tipo = "perro";

console.log(perro.describir()); // "Soy un perro llamado Toby"

// Crear con propiedades propias
const gato2 = Object.create(prototipoAnimal, {
  nombre: { value: "Michi", writable: true },
  tipo: { value: "gato" }
});

Verificando la Cadena

// hasOwnProperty vs in
console.log(gato.hasOwnProperty("nombre")); // true (propia)
console.log(gato.hasOwnProperty("saludar")); // false (del prototype)
console.log("nombre" in gato);  // true (propia o heredada)
console.log("saludar" in gato); // true (del prototype)

// Obtener el prototipo
console.log(Object.getPrototypeOf(gato) === Mascota.prototype); // true

✗ Incorrecto

// Modificar prototype de Object (peligroso)

// Afecta TODOS los objetos!
// for...in enumerara el metodo nuevo

✓ Correcto

  return "Soy " + this.nombre + ", tipo " + this.tipo;

Cuidado: Nunca modifiques Object.prototype directamente. Esto afecta a todos los objetos de JavaScript y puede causar bugs extremadamente dificiles de depurar.

Ejemplo Practico: Calculo de Distancia

function calcularDistancia(lat1, lon1, lat2, lon2) {
  const R = 6371; // Radio de la Tierra en km
  const dLat = (lat2 - lat1) * Math.PI / 180;
  const dLon = (lon2 - lon1) * Math.PI / 180;
  const a =
    Math.sin(dLat/2) ** 2 +
    Math.cos(lat1 * Math.PI/180) *
    Math.cos(lat2 * Math.PI/180) *
    Math.sin(dLon/2) ** 2;
  const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));
  return R * c; // Distancia en km
}

8.6 Intersection Observer API

Intersection Observer permite observar cuando un elemento entra o sale del viewport (o de un contenedor padre). Es extremadamente util para lazy loading, infinite scroll y animaciones al scroll.

Uso Basico

const observer = new IntersectionObserver(
  (entries) => {
    entries.forEach(entry => {
      if (entry.isIntersecting) {
        console.log("Elemento visible:", entry.target.id);
      }
    });
  },
  {
    root: null,       // viewport
    rootMargin: "0px",
    threshold: 0.5  // 50% visible
  }
);

// Observar un elemento
const elemento = document.getElementById("mi-elemento");
observer.observe(elemento);

// Dejar de observar
observer.unobserve(elemento);

Lazy Loading de Imagenes

const imagenes = document.querySelectorAll("img[data-src]");

const lazyObserver = new IntersectionObserver((entries) => {
  entries.forEach(entry => {
    if (entry.isIntersecting) {
      const img = entry.target;
      img.src = img.dataset.src; // Cargar imagen real
      img.removeAttribute("data-src");
      lazyObserver.unobserve(img);
    }
  });
}, { rootMargin: "200px" }); // Cargar 200px antes

imagenes.forEach(img => lazyObserver.observe(img));

Infinite Scroll

const sentinel = document.getElementById("sentinel");
let pagina = 1;

const scrollObserver = new IntersectionObserver(async (entries) => {
  if (entries[0].isIntersecting) {
    scrollObserver.unobserve(sentinel);

    await cargarMasPost(pagina);

class Router {
  #rutas = {};

  agregarRuta(path, handler) {
    this.#rutas[path] = handler;

    this.ejecutar(path);

    if (handler) handler();

    window.addEventListener("popstate", (e) => {
      this.ejecutar(e.state?.path || "/");
    });
    this.ejecutar(location.pathname);
  }
}

9.3 ES Modules vs CommonJS

JavaScript tiene dos sistemas de modulos principales: ES Modules (import/export, estandar moderno) y CommonJS (require/module.exports, usado en Node.js legacy). Entender ambas es esencial.

ES Modules (import/export)

// math.js - Exportar funciones
export const PI = 3.14159;

export function sumar(a, b) {
  return a + b;
}

export function restar(a, b) {
  return a - b;
}

// app.js - Importar
import { sumar, restar, PI } from "./math.js";

console.log(sumar(5, 3)); // 8
console.log(PI);         // 3.14159

// Importar todo el modulo
import * as math from "./math.js";
console.log(math.sumar(1, 2));

CommonJS (require/module.exports)

// utils.js - CommonJS (Node.js)
function formatearFecha(fecha) {
  return fecha.toLocaleDateString("es-ES");
}

module.exports = { formatearFecha };

// server.js - Importar
const { formatearFecha } = require("./utils");
console.log(formatearFecha(new Date()));

// Cargar modulos bajo demanda (code splitting)

10.4 Sistema de Rutas

Un SPA (Single Page Application) necesita un sistema de rutas que permita navegar sin recargar la pagina. Usaremos hash routing, la forma mas simple y compatible.

Router Basico con Hash

// router.js - Sistema de rutas basado en hash
const rutas = {};
let rutaActual = null;

export function definirRuta(hash, callback) {
  rutas[hash] = callback;
}

export function navegarA(hash) {
  window.location.hash = hash;
}

function manejarRuta() {
  const hash = window.location.hash || "#/";

  if (rutaActual === hash) return;
  rutaActual = hash;

  const callback = rutas[hash];

  if (callback) {
    callback();
  } else {
    mostrar404();
  }
}

window.addEventListener("hashchange", manejarRuta);
window.addEventListener("load", manejarRuta);

10.7 Estado y Almacenamiento

Manejar el estado de la aplicacion de forma centralizada evita bugs y hace el codigo mas predecible. Combinaremos localStorage con un patron de estado simple.

Wrapper de localStorage

// services/storage.js
export const storage = {
  obtener(clave, valorDefecto = null) {
    try {
      const dato = localStorage.getItem(clave);
      return dato ? JSON.parse(dato) : valorDefecto;
    } catch (e) {
      console.error("Error leyendo de localStorage:", e);
      return valorDefecto;
    }
  },

  guardar(clave, valor) {
    try {
      localStorage.setItem(clave, JSON.stringify(valor));
      return true;
    } catch (e) {
      console.error("Error guardando en localStorage:", e);
      return false;
    }
  },

  eliminar(clave) {
    localStorage.removeItem(clave);
  }
};

Patron de Estado con Observadores

// state.js - Estado centralizado con suscriptores
import { storage } from "./services/storage.js";

const suscriptores = [];

let estado = {
  usuario: storage.obtener("usuario"),
  tareas: storage.obtener("tareas", []),
  filtro: "todas",
  busqueda: ""
};

export function obtenerEstado() {
  return { ...estado };
}

export function actualizarEstado(cambios) {
  estado = { ...estado, ...cambios };

  // Persistir datos importantes
  if (cambios.tareas) {
    storage.guardar("tareas", estado.tareas);
  }
  if (cambios.usuario) {
    storage.guardar("usuario", estado.usuario);
  }

  // Notificar a todos los suscriptores
  suscriptores.forEach((fn) => fn(estado));
}

export function suscribirse(callback) {
  suscriptores.push(callback);
  // Retornar funcion para cancelar suscripcion
  return () => {
    const idx = suscriptores.indexOf(callback);
    if (idx > -1) suscriptores.splice(idx, 1);
  };
}

Patron observable: Cuando el estado cambia, todos los suscriptores son notificados automaticamente. Esto permite que la interfaz se actualice sin acoplamiento directo.

Busqueda en Tiempo Real

// Busqueda con debounce para no sobrecargar el DOM
function debounce(funcion, espera = 300) {
  let temporizador;
  return function(...args) {
    clearTimeout(temporizador);
    temporizador = setTimeout(() => funcion.apply(this, args), espera);
  };
}

function filtrarTareas(tareas, termino) {
  const busqueda = termino.toLowerCase().trim();
  if (!busqueda) return tareas;

  return tareas.filter((t) =>
    t.titulo.toLowerCase().includes(busqueda)
  );
}

// Aplicar debounce al input de busqueda
const inputBusqueda = document.getElementById("search-input");
inputBusqueda.addEventListener("input",
  debounce((e) => {
    const resultado = filtrarTareas(
      obtenerEstado().tareas,
      e.target.value
    );
    renderTaskList(document.getElementById("app"), resultado);
  })
);

// Paginacion de resultados
function paginar(items, paginaActual, elementosPorPagina = 10) {
  const inicio = (paginaActual - 1) * elementosPorPagina;
  const fin = inicio + elementosPorPagina;

  return {
    items: items.slice(inicio, fin),
    totalPaginas: Math.ceil(items.length / elementosPorPagina),
    paginaActual: paginaActual,
    tieneSiguiente: fin < items.length,
    tieneAnterior: paginaActual > 1
  };
}

// Renderizar controles de paginacion
function renderPaginacion(container, info) {
  container.innerHTML = `
    <nav class="pagination" aria-label="Paginacion">
      <button class="btn-page" ${!info.tieneAnterior ? 'disabled' : ''}
        data-page="${info.paginaActual - 1}">Anterior</button>
      <span class="page-info">
        Pagina ${info.paginaActual} de ${info.totalPaginas}
      </span>
      <button class="btn-page" ${!info.tieneSiguiente ? 'disabled' : ''}
        data-page="${info.paginaActual + 1}">Siguiente</button>
    </nav>
  `;
}

Ordenamiento

// Ordenar tareas por diferentes criterios
function ordenarTareas(tareas, criterio, direccion = "asc") {
  const copia = [...tareas];

  copia.sort((a, b) => {
    let comparacion = 0;

    switch (criterio) {
      case "fecha":
        comparacion = new Date(a.fecha) - new Date(b.fecha);
        break;
      case "nombre":
        comparacion = a.titulo.localeCompare(b.titulo, "es");
        break;
      case "estado":
        comparacion = a.completada - b.completada;
        break;
    }

    return direccion === "desc" ? -comparacion : comparacion;
  });

  return copia;
}

// Uso
const tareasOrdenadas = ordenarTareas(
  obtenerEstado().tareas,
  "fecha",
  "desc"
);
